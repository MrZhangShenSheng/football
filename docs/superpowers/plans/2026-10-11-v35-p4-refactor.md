# engine/scripts/surviving_chain.py
# -*- coding: utf-8 -*-
"""v35-P4 D6 幸存链注册表——生产组件白名单+PASS 凭证（铁律15 机器闸门）。

无审计不入链：每个生产概率源组件在 surviving_chain.json 挂凭证（unit+CI+审计日期）；
核心组件启动校验，缺/过期/非 keep → 拒跑（SystemExit 2）。
fail-open 只适用仪表（evidence hook）；闸门是准入不是仪表，拒跑是正道。
开发者 sszhang
"""
import json
from datetime import date
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]
REGISTRY_PATH = ROOT / "engine" / "cache" / "strength_chain" / "surviving_chain.json"

STATUS_KEEP = "keep"          # 留任主概率源（可作启动闸校验对象）
STATUS_DEMOTE = "demote"      # 降级层/调味（不得作主干）
STATUS_RETIRED = "retired"    # 退场（历史档案）
MAX_AUDIT_AGE_DAYS = 90       # 凭证有效期


def load_registry() -> dict:
    """缺文件/脏文件返回空注册表（由 check_component 给出拒跑原因，不在这里抛）。"""
    try:
        return json.loads(REGISTRY_PATH.read_text(encoding="utf-8"))
    except Exception:                      # noqa: BLE001
        return {"schema": 1, "components": {}}


def check_component(name: str, max_age_days: int = MAX_AUDIT_AGE_DAYS) -> tuple:
    """(ok, reason)。ok=True 仅当：条目存在 + status==keep + 审计未过期。"""
    reg = load_registry()
    comp = (reg.get("components") or {}).get(name)
    if not comp:
        return False, f"[surviving_chain] 组件 {name} 不在注册表——无审计不入链（铁律15），先过消融审计"
    if comp.get("status") != STATUS_KEEP:
        return False, f"[surviving_chain] 组件 {name} status={comp.get('status')} 非 keep——不得作主干"
    audited = str(comp.get("auditedAt") or "")
    try:
        age = (date.today() - date.fromisoformat(audited)).days
    except ValueError:
        return False, f"[surviving_chain] 组件 {name} auditedAt 非法: {audited!r}"
    if age > max_age_days:
        return False, f"[surviving_chain] 组件 {name} 审计过期（{audited}·{age}天>{max_age_days}）——重审后才可跑"
    return True, f"PASS · {name} 凭证 {audited}（{age}天）"


def require_component(name: str) -> None:
    """启动闸：不通过直接 SystemExit(2)——防未审计组件混入生产。"""
    ok, why = check_component(name)
    if not ok:
        raise SystemExit(why)
    print(f"[surviving_chain] {why}")
```

"""市场反解引擎（v35-P4·D3）——体彩盘口直接反解概率/λ·零模型零画像依赖。

证据（surviving_chain.json 凭证·2026-10-11）:
  crs_power_devig  <- U2+铁律13: 市场top1 13.5%/top3 33.5% = 现存最好的比分分布
  market_lambda_solver <- U2: 矩法反解λ双null显著胜V3W(Δ+0.0994/+0.2488)·ρ=0(U4)
开发者 sszhang
"""
from pathlib import Path
import sys

import numpy as np

sys.path.insert(0, str(Path(__file__).resolve().parent))
import score_matrix as sm          # 同目录·dc_matrix/had_from 口径同链

MIN_CRS_CELLS = 20                 # 盘口<20项→降级（crs_fusion marketFused 门槛沿袭）
RHO_ZERO = 0.0                     # U4: ρ 修正判死·独立泊松
TTG_KEYS = tuple(str(i) for i in range(8))   # 体彩总进球 0..7（"7"=7+ 档）
TTG_PLUS_VAL = 7.0                 # "7+"期望替身·保守下界（p7 通常<1%）
CRS_OTHER_MAP = {"other_h": "s1sh", "other_d": "s1sd", "other_a": "s1sa"}


def crs_key(k: str) -> str:
    """市场盘键→39格矩阵键：'1:1'→'s01s01'·other 三键→s1sx·非法返回 ''。"""
    k = str(k)
    if k in CRS_OTHER_MAP:
        return CRS_OTHER_MAP[k]
    if ":" in k:
        h, a = k.split(":", 1)
        if h.isdigit() and a.isdigit() and int(h) <= 5 and int(a) <= 5:
            return f"s{int(h):02d}s{int(a):02d}"
    return ""


def _power_devig_probs(inv: np.ndarray) -> np.ndarray:
    """k 二分解 Σ(1/o)^k=1（U3 修复版目标·2026-10-11 翻案实现）。"""
    lo, hi = 0.5, 5.0
    for _ in range(40):
        mid = (lo + hi) / 2
        lo, hi = (mid, hi) if (inv ** mid).sum() > 1 else (lo, mid)
    p = inv ** ((lo + hi) / 2)
    return p / p.sum()


def crs_power_devig(crs: dict) -> dict:
    """体彩 CRS 31 格 power 去水→sXXsYY 键分布。盘口不足返回 {}（由上层降级）。"""
    cells = {}
    for k, v in (crs or {}).items():
        try:
            o = float(v)
        except (TypeError, ValueError):
            continue
        ck = crs_key(k)
        if ck and o > 1.0:
            cells[ck] = o
    if len(cells) < MIN_CRS_CELLS:
        return {}
    p = _power_devig_probs(np.array(list(cells.values())))
    return {k: float(x) for k, x in zip(cells.keys(), p)}


def had_devig(o_h, o_d, o_a) -> list:
    inv = np.array([1 / float(o_h), 1 / float(o_d), 1 / float(o_a)])
    return [float(x) for x in inv / inv.sum()]


def lam_sum_from_ttg(ttg: dict) -> float:
    """E[总进球]：TTG power 去水后期望（确定性无优化·预注册造法）。"""
    ks = sorted((str(k) for k in ttg if str(k) in TTG_KEYS and float(ttg[k]) > 1.0),
                key=float)
    p = _power_devig_probs(np.array([1.0 / float(ttg[k]) for k in ks]))
    return float(sum((TTG_PLUS_VAL if k == "7" else float(k)) * pi for k, pi in zip(ks, p)))


def ttg_boom(ttg: dict) -> float:
    """市场口径 boom：TTG 6+7 档去水概率合计（替代 V3W other 桶 boom 语义）。"""
    ks = sorted((str(k) for k in ttg if str(k) in TTG_KEYS and float(ttg[k]) > 1.0),
                key=float)
    p = _power_devig_probs(np.array([1.0 / float(ttg[k]) for k in ks]))
    return float(sum(pi for k, pi in zip(ks, p) if k in ("6", "7")))


def solve_lamh(lam_sum: float, p_h: float) -> float:
    """40 次二分解 P(H>A)(λh, λsum−λh)=p_h（had_from 聚合·ρ=0·λh∈(0.05, λsum−0.05)）。"""
    lo, hi = 0.05, max(0.06, lam_sum - 0.05)
    for _ in range(40):
        mid = (lo + hi) / 2
        p = sm.had_from(sm.dc_matrix(mid, lam_sum - mid, RHO_ZERO))["h"]
        lo, hi = (mid, hi) if p < p_h else (lo, mid)
    return (lo + hi) / 2
```

def load_day_odds(day: str) -> dict:
    """扫 score_odds 相邻三日存档·按块 businessDate==day 合并·键=matchNumStr（存档无 matchId）。

    存档文件日=抓取日≠比赛日（一日存档含多日块）——开赛日口径铁律。
    """
    from datetime import date as _d, timedelta as _td
    d0 = _d.fromisoformat(day)
    out = {}
    for off in (-1, 0, 1):
        f = ROOT / "engine" / "cache" / "score_odds" / f"{(d0 + _td(days=off)).isoformat()}.json"
        if not f.exists():
            continue
        try:
            doc = json.loads(f.read_text(encoding="utf-8"))
        except Exception:                  # noqa: BLE001
            continue
        for blk in doc.get("matchDays") or []:
            if str(blk.get("businessDate")) != day:
                continue
            for m in blk.get("matches") or []:
                if m.get("matchNumStr"):
                    out[str(m["matchNumStr"])] = m
    return out


def pick_source(odds: dict) -> dict or None:
    """三层降级前两层（市场侧·纯函数）：层1 CRS去水主干 → 层2 反解λ矩阵(ρ=0)。None=走层3 V3W。"""
    crs_p = mkl.crs_power_devig(odds.get("crs") or {})
    if crs_p:
        top5 = sorted(crs_p.items(), key=lambda kv: -kv[1])[:5]
        return {"src": "market_crs", "lam": None,
                "top5": [{"pick": k, "p": round(v, 4)} for k, v in top5],
                "boom": mkl.ttg_boom(odds.get("ttg") or {})}
    ttg, had = odds.get("ttg") or {}, odds.get("had") or {}
    if len(ttg) >= 6 and had.get("h") and had.get("d") and had.get("a"):
        lam_sum = mkl.lam_sum_from_ttg(ttg)
        p3 = mkl.had_devig(had["h"], had["d"], had["a"])
        lh = mkl.solve_lamh(lam_sum, p3[0])
        mat = sm.dc_matrix(lh, lam_sum - lh, mkl.RHO_ZERO)     # D4: ρ=0
        top5 = sorted(mat.items(), key=lambda kv: -kv[1])[:5]
        boom = sum(mat.get(k, 0.0) for k in ("s1sh", "s1sd", "s1sa"))
        return {"src": "market_lambda", "lam": [round(lh, 3), round(lam_sum - lh, 3)],
                "top5": [{"pick": k, "p": round(v, 4)} for k, v in top5],
                "boom": round(boom, 4)}
    return None
