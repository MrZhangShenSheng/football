def _league_rows(league_dir: str) -> list[dict]:
    """league 库行（date/home/away/hg/ag）带缓存。开发者 sszhang"""
    if league_dir not in _league_cache:
        path = LEAGUE_DIR / f"{league_dir}_matches.json"
        rows = json.loads(path.read_text(encoding="utf-8")) if path.exists() else []
        _league_cache[league_dir] = rows if isinstance(rows, list) else rows.get("matches", [])
    return _league_cache[league_dir]

def fd_date(d: str) -> str:
    """fd dd/mm/YYYY → ISO。开发者 sszhang"""
    dd, mm, yy = d.split("/")
    return f"{yy}-{mm.zfill(2)}-{dd.zfill(2)}"

def _match_teams(iso: str, hg: int, ag: int, league_dir: str) -> tuple[str | None, str | None]:
    """三键匹配（日期±1+比分）→ 规范 ID (home, away)。多候选取日期最近。开发者 sszhang"""
    d0 = date.fromisoformat(iso)
    best = None
    for r in _league_rows(league_dir):
        if r["hg"] != hg or r["ag"] != ag:
            continue
        gap = abs((date.fromisoformat(r["date"][:10]) - d0).days)
        if gap <= 1 and (best is None or gap < best[0]):
            best = (gap, r["home"], r["away"])
    return (best[1], best[2]) if best else (None, None)

def _f(v: str) -> float | None:
    v = (v or "").strip()
    return float(v) if v else None

def hist_match_record(row: dict, league_dir: str, league_zh: str) -> dict:
    """fd CSV 行 → MatchRecord 历史口径。odds=收盘 PSCH 优先缺则 B365 开盘（evidence 标）。开发者 sszhang"""
    iso = fd_date(row["Date"])
    hg, ag = int(row["FTHG"]), int(row["FTAG"])
    th, ta = _match_teams(iso, hg, ag, league_dir)
    odds = None
    if all(_f(row.get(f"PSC{k}")) for k in "HDA"):
        odds = [_f(row[f"PSC{k}"]) for k in "HDA"]
    elif all(_f(row.get(f"B365{k}")) for k in "HDA"):
        odds = [_f(row[f"B365{k}"]) for k in "HDA"]
    return {
        "date": iso, "kickoff": iso, "code": f"hist:{iso}", "league": league_zh,
        "match": f"{row['HomeTeam']} vs {row['AwayTeam']}",
        "team_h": th, "team_a": ta, "league_dir": league_dir,
        "odds": odds, "pin_close": None, "elo_h": None, "elo_a": None,
        "inj_n_h": 0, "inj_n_a": 0, "rest_days_h": None, "rest_days_a": None,
        "result": {"hg": hg, "ag": ag,
                   "ht_hg": _f(row.get("HTHG")), "ht_ag": _f(row.get("HTAG"))},
        "red_h": int(_f(row.get("HR")) or 0), "red_a": int(_f(row.get("AR")) or 0),
        "hst": int(_f(row.get("HST")) or 0), "ast": int(_f(row.get("AST")) or 0),
        "xg_h": _f(row.get("HxG")), "xg_a": _f(row.get("AxG")),
    }

# attribution_dims.py 追加（新注释节 # ── 期2 事件维度 ── 下）
RED_CARD_GOAL = 0.35   # 每张红牌给对方进球期望的偏移（先验·待频次表校准）
SHOT_CONV_BASE = 0.30  # 射正→进球联赛基准转化率（先验·待频次表校准）
FINISH_SCALE = 0.5     # xG 背离入账比例（先验·待频次表校准）

def dim_red(rec: dict) -> dict | None:
    """E1 红牌（fd 无分钟时点·张数近似）：得牌方对方 +RED_CARD_GOAL×张。开发者 sszhang"""
    rh, ra = rec.get("red_h", 0), rec.get("red_a", 0)
    if rh == 0 and ra == 0:
        return None
    return {"dim": "E1.red", "h": round(RED_CARD_GOAL * ra, 3), "a": round(RED_CARD_GOAL * rh, 3),
            "evidence": f"红牌 h{rh}/a{ra}(无时点·张数近似)"}

def dim_shots(rec: dict) -> dict | None:
    """E4 射正背离：转化率差×射正数（正=转化超额侧加分）。缺射正返回 None。开发者 sszhang"""
    hst, ast = rec.get("hst", 0), rec.get("ast", 0)
    r = rec.get("result") or {}
    if not hst and not ast:
        return None
    hg, ag = r.get("hg") or 0, r.get("ag") or 0
    conv_h = hg / hst if hst else SHOT_CONV_BASE
    conv_a = ag / ast if ast else SHOT_CONV_BASE
    return {"dim": "E4.shots",
            "h": round((conv_h - SHOT_CONV_BASE) * hst, 3),
            "a": round((conv_a - SHOT_CONV_BASE) * ast, 3),
            "evidence": f"射正转化 h{conv_h:.2f}/a{conv_a:.2f} vs 基准{SHOT_CONV_BASE}"}

def dim_finishing(rec: dict) -> dict | None:
    """E5 把握背离（xG 双侧齐）：实际进球−xG 按 FINISH_SCALE 入账。开发者 sszhang"""
    xh, xa = rec.get("xg_h"), rec.get("xg_a")
    if xh is None or xa is None:
        return None
    r = rec.get("result") or {}
    return {"dim": "E5.finishing",
            "h": round(((r.get("hg") or 0) - xh) * FINISH_SCALE, 3),
            "a": round(((r.get("ag") or 0) - xa) * FINISH_SCALE, 3),
            "evidence": f"xG {xh}:{xa} vs 实际 {r.get('hg')}:{r.get('ag')}"}

# attribution_hist.py 追加
import sys as _sys
_sys.path.insert(0, str(SCRIPTS_DIR))
from score_attribution import build_bill

def run_batch(season: str, codes: list[str]) -> None:
    """60 CSV（或指定子集）→ hist/{code}_{season}-attribution.json。开发者 sszhang"""
    out_dir = ROOT / "data" / "04-summaries" / "score-attribution" / "hist"
    out_dir.mkdir(parents=True, exist_ok=True)
    for code in codes:
        league_dir, league_zh = FD_LEAGUES[code]
        path = FD_DIR_PATH / f"{code}_{season}.csv"
        if not path.exists():
            print(f"[跳过] {code}_{season} 无文件"); continue
        rows = list(csv.DictReader(open(path, encoding="latin-1")))
        bills, unmatched, no_odds = [], 0, 0
        for row in rows:
            if not (row.get("FTHG") or "").strip():
                continue
            rec = hist_match_record(row, league_dir, league_zh)
            if rec["team_h"] is None:
                unmatched += 1
            if rec["odds"] is None:
                no_odds += 1
            bills.append(build_bill(rec))
        out = out_dir / f"{code}_{season}-attribution.json"
        out.write_text(json.dumps(bills, ensure_ascii=False), encoding="utf-8")
        ok = len([b for b in bills if "baseline" in b])
        print(f"{code}_{season}: {len(rows)}行 出账{ok} 队名未匹配{unmatched} odds缺{no_odds}")

def load_bills() -> list[dict]:
    """hist/ 全部账本加载（forward 账本期3并入）。开发者 sszhang"""
    bills = []
    for p in sorted(HIST_DIR.glob("*-attribution.json")):
        bills.extend(json.loads(p.read_text(encoding="utf-8")))
    return [b for b in bills if "baseline" in b]

def _wilson_lo(k: int, n: int) -> float:
    """Wilson 95% 下限。开发者 sszhang"""
    if n == 0:
        return 0.0
    z, p = 1.96, k / n
    denom = 1 + z * z / n
    center = p + z * z / (2 * n)
    rad = z * math.sqrt(p * (1 - p) / n + z * z / (4 * n * n))
    return max(0.0, (center - rad) / denom)

def freq_table(bills: list[dict], on, band) -> dict:
    """单表：on 组 vs off 组的 band 命中率差（Wilson CI）。开发者 sszhang"""
    on_b = [b for b in bills if on(b)]; off_b = [b for b in bills if not on(b)]
    k_on = sum(1 for b in on_b if band(b)); k_off = sum(1 for b in off_b if band(b))
    n_on, n_off = len(on_b), len(off_b)
    return {"n_on": n_on, "n_off": n_off,
            "rate_on": round(k_on / n_on, 4) if n_on else None,
            "rate_off": round(k_off / n_off, 4) if n_off else None,
            "diff": round((k_on / n_on - k_off / n_off), 4) if n_on and n_off else None,
            "ci_lo_on": round(_wilson_lo(k_on, n_on), 4) if n_on else None}
