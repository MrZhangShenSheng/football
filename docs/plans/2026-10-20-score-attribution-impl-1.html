"""比分归因引擎·期1：forward 卷适配 + 账单组装 + CLI。
设计 docs/plans/2026-10-20-score-attribution-design.html。开发者 sszhang"""
import json
import sys
from datetime import date, datetime
from pathlib import Path

SCRIPTS_DIR = Path(__file__).resolve().parent
sys.path.insert(0, str(SCRIPTS_DIR))

import strength_loaders as sl  # load_aliases/zh_to_id 复用

ROOT = SCRIPTS_DIR.parents[1]
RESULTS_DIR = ROOT / "data" / "02-results"
LINEUPS_DIR = ROOT / "data" / "05-trends"
TEAMS_DIR = ROOT / "data" / "01-teams"


def _parse_score(s: str | None) -> tuple[int, int] | None:
    """'4-0' → (4,0)；'1:0'（半场）同构。失败返回 None。开发者 sszhang"""
    if not s:
        return None
    for sep in ("-", ":"):
        if sep in s:
            a, b = s.split(sep)[:2]
            return int(a), int(b)
    return None


def _find_in_results(date_str: str, code: str) -> dict | None:
    """02-results/{date}.json 按 code 找 match 原始行。开发者 sszhang"""
    path = RESULTS_DIR / f"{date_str}.json"
    if not path.exists():
        return None
    data = json.loads(path.read_text(encoding="utf-8"))
    for m in data.get("matches", []):
        if m.get("code") == code:
            return m
    return None


def _team_profile(team_id: str | None) -> dict:
    """规范ID→画像 dict（elo/standing/form）；无档案返回 {}。开发者 sszhang"""
    if not team_id:
        return {}
    hits = list(TEAMS_DIR.glob(f"*/{team_id}.json"))
    if not hits:
        return {}
    return json.loads(hits[0].read_text(encoding="utf-8"))


def _injury_count(date_str: str, code: str) -> tuple[int, int]:
    """lineup 轨最后快照伤停人数 (h,a)；无轨返回 (0,0)。开发者 sszhang"""
    path = LINEUPS_DIR / f"{date_str}-lineups.json"
    if not path.exists():
        return 0, 0
    data = json.loads(path.read_text(encoding="utf-8"))
    rec = (data.get("matches") or {}).get(code) or {}
    snaps = rec.get("injSnapshots") or []
    if not snaps:
        return 0, 0
    inj = snaps[-1].get("inj") or {}
    return len(inj.get("h") or []), len(inj.get("a") or [])


def _last_match_date(team_id: str, league_dir: str, before: str) -> str | None:
    """league 库该队 before 前最近一场日期（算休息天数用）。开发者 sszhang"""
    path = RESULTS_DIR / "league" / f"{league_dir}_matches.json"
    if not path.exists() or not team_id:
        return None
    rows = json.loads(path.read_text(encoding="utf-8"))
    rows = rows if isinstance(rows, list) else rows.get("matches", [])
    cands = [r["date"] for r in rows
             if r["date"] < before and team_id in (r.get("home"), r.get("away"))]
    return max(cands) if cands else None


def load_match_record(date_str: str, code: str) -> dict | None:
    """forward 卷单场 → MatchRecord（账单引擎唯一输入）。开发者 sszhang"""
    m = _find_in_results(date_str, code)
    if m is None:
        return None
    zh_map = sl.zh_to_id()
    home_zh, away_zh = m["match"].split(" vs ")
    team_h, team_a = zh_map.get(home_zh), zh_map.get(away_zh)
    prof_h, prof_a = _team_profile(team_h), _team_profile(team_a)
    hg, ag = _parse_score(m.get("result")) or (None, None)
    ht_hg, ht_ag = _parse_score(m.get("half")) or (None, None)
    inj_h, inj_a = _injury_count(date_str, code)
    o = m.get("odds") or {}
    league_dir = LEAGUE_DIR_MAP.get(m.get("league"), "")
    return {
        "date": date_str, "code": code, "league": m.get("league"),
        "match": m["match"], "team_h": team_h, "team_a": team_a,
        "league_dir": league_dir,
        "odds": [float(o["h"]), float(o["d"]), float(o["a"])],
        "pin_close": m.get("pinClose") or None,
        "elo_h": prof_h.get("elo"), "elo_a": prof_a.get("elo"),
        "inj_n_h": inj_h, "inj_n_a": inj_a,
        "rest_days_h": None, "rest_days_a": None,   # Task 5 填
        "result": {"hg": hg, "ag": ag, "ht_hg": ht_hg, "ht_ag": ht_ag},
    }

"""比分归因维度计算器（纯函数·同输入同输出）。设计 §二。开发者 sszhang"""
import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parent))
from market_lambda import had_devig, solve_lamh


def lambda_from_had(odds: list[float], lam_sum: float) -> tuple[float, float] | None:
    """L1 基线：HAD 三向去水→p_h→solve_lamh 反解 (λh, λa)。开发者 sszhang"""
    if lam_sum is None or lam_sum <= 0.5:
        return None
    p_h, _, _ = had_devig(*odds)
    if not 0.05 < p_h < 0.95:
        return None
    lam_h = solve_lamh(lam_sum, p_h)
    return round(lam_h, 3), round(lam_sum - lam_h, 3)

ELO_SCALE = 400.0  # Elo 差 400≈净胜 1 球（经验先验·待频次表校准·设计§二A1）

def dim_elo(elo_h, elo_a) -> dict | None:
    """A1 实力差→净胜偏移（对半分给 h/a）。缺 Elo 返回 None 进 missing。开发者 sszhang"""
    if elo_h is None or elo_a is None:
        return None
    net = (elo_h - elo_a) / ELO_SCALE
    return {"dim": "A1.elo", "h": round(net / 2, 3), "a": round(-net / 2, 3),
            "evidence": f"elo_diff={round(elo_h - elo_a)}"}

# attribution_dims.py 追加
import json

FORMATS_PATH = Path(__file__).resolve().parents[1] / "data" / "00-leagues" / "_formats.json"
FIFA_VIRUS_DAYS = 5  # 窗口结束后 N 天内的联赛场算 FIFA 病毒（先验·待频次表校准）

def _formats() -> dict:
    return json.loads(FORMATS_PATH.read_text(encoding="utf-8"))

def _in_fifa_window_tail(date_str: str) -> bool:
    for win in _formats().get("_fifa_windows_2026", []):
        end = win.split("/")[1]
        d0 = datetime.fromisoformat(date_str).date()
        d1 = datetime.fromisoformat(end).date()
        if 0 < (d0 - d1).days <= FIFA_VIRUS_DAYS:
            return True
    return False

def dim_stage(rec: dict) -> dict:
    """L0 赛制情境→形状系数（期1只判 league_regular/fifa_virus；淘汰赛档期4接 C1 细分）。开发者 sszhang"""
    fmt = _formats().get(rec.get("league"))
    if fmt is None:
        return {"dim": "L0.stage", "tail_scale": 1.0, "evidence": "no_format"}
    ev = "league_regular" if fmt.get("type", "league").startswith("league") else fmt["type"]
    if _in_fifa_window_tail(rec["date"]):
        ev += "+fifa_virus"
    return {"dim": "L0.stage", "tail_scale": 1.0, "evidence": ev}

HTFT_SURGE_PER_GOAL = 0.35  # 下半场净增每球的偏移贡献（先验·待频次表校准）

def dim_htft(rec: dict) -> dict | None:
    """E2 半场→全场结构：下半场净增球的归属偏移。缺半场比分返回 None。开发者 sszhang"""
    r = rec.get("result") or {}
    if r.get("ht_hg") is None or r.get("hg") is None:
        return None
    s2_h = r["hg"] - r["ht_hg"]
    s2_a = r["ag"] - r["ht_ag"]
    net = s2_h - s2_a
    half = net * HTFT_SURGE_PER_GOAL / 2
    return {"dim": "E2.htft", "h": round(half, 3), "a": round(-half, 3),
            "evidence": f"半场{r['ht_hg']}:{r['ht_ag']}→全场{r['hg']}:{r['ag']}"}

def dim_injury_coarse(rec: dict) -> dict | None:
    """R1 粗档（期1标记不量化；LLM 轨量化在期3）。人数差≥2 才标记。开发者 sszhang"""
    nh, na = rec.get("inj_n_h", 0), rec.get("inj_n_a", 0)
    if abs(nh - na) < 2:
        return None
    return {"dim": "R1.coarse", "h": None, "a": None,
            "evidence": f"伤停人数差 h{nh}/a{na}(未量化)"}

def classify_script(rec: dict) -> dict:
    """L5 剧本规则定档（期1可判6/8：红牌/门将/默契待期2画面数据）。开发者 sszhang"""
    r = rec.get("result") or {}
    hg, ag = r.get("hg"), r.get("ag")
    if hg is None:
        return {"type": "unclassified", "reason": "no_result"}
    hth, hta = r.get("ht_hg"), r.get("ht_ag")
    net, tot = hg - ag, hg + ag
    if net >= 3:
        if hth is not None and (hth - hta) >= 1:
            return {"type": "crush", "reason": f"净胜{net}·半场已领先{hth}:{hta}"}
        return {"type": "early_crush", "reason": f"净胜{net}·半场{hth}:{hta}"}
    if net == 0 and tot <= 1:
        return {"type": "iron_grid", "reason": f"总球{tot}互磨"}
    if tot >= 5:
        return {"type": "leaky", "reason": f"总球{tot}互漏"}
    if hth is not None:
        s2_net = (hg - hth) - (ag - hta)
        if hth != hta and s2_net * (hth - hta) > 0 and abs(s2_net) >= 2:
            return {"type": "press_counter", "reason": "半场领先方下半场继续扩大=压上反杀族"}
        if abs(net) <= 1 and s2_net != 0:
            return {"type": "open_play", "reason": "开放对攻有来有回"}
    if tot <= 1:
        return {"type": "early_goal_lock", "reason": f"总球{tot}"}
    return {"type": "open_play", "reason": "默认开放"}

def league_avg_total(league: str) -> tuple[float, str]:
    """00-leagues 画像 standings 算场均总球（Σgf/Σplayed×2，两队合计口径）；读不到用默认。开发者 sszhang"""
    path = ROOT / "data" / "00-leagues" / f"{LEAGUE_DIR_MAP.get(league, '')}.json"
    try:
        data = json.loads(path.read_text(encoding="utf-8"))
        gf = sum(s.get("gf", 0) for s in data.get("standings", []))
        played = sum(s.get("played", 0) for s in data.get("standings", []))
        if gf and played:
            return round(2 * gf / played, 2), "league_profile"
    except Exception:
        pass
    return LEAGUE_AVG_DEFAULT, "default"
