一、核心架构决策（执行代理必读）

1. 特征无泄漏原则（本计划最重要的一条）

球队画像（data/01-teams/）是当前时点快照、联赛 standings 是全季聚合——两者对历史逐场回测都有时间错位泄漏。因此阶段0 特征全部从 engine/cache/odds_{league}_{season}.json 的场次序列自身滚动构造：每场只用该场之前的比分数据算特征。

2. 阶段0 特征集（3 项，全滚动）与超参（手写，待阶段1学习器替换）

特征构造作用面
F1 攻击率主队近5主场场均进球（小样本向联赛主队侧均值收缩）÷ 联赛主队侧均值 → ratio_att_h；对称得客队λ 同侧乘子
F2 防守率对侧近5客场场均失球（同样收缩）÷ 联赛客队侧均值 → ratio_def；对称得主队λ 对侧乘子
F3 开季降权该队此前场次 <2 时，特征权重 ×0.3（样本不足回落裸 DC；语义=早期收缩，非硬乘 λ）特征层
λ_home' = λ_home_DC × exp( β·ln(ratio_att_h) + β·ln(ratio_def_a) )，β=0.3；λ_away' 对称
小样本收缩：rate = (队内累计 + 联赛均值×2.0) / (场次 + 2.0)

β=0.3 与收缩先验 k=2.0 是阶段0 手写超参——报告里必须标注「待阶段1学习器替换+消融验证」。消融开关三个：att / def / shrink。

3. 朴素基线双口径（复审 A3 落地）

naiveStatic＝全季最常见固定档（TTG 取最常见 3 档、CRS 取最常见 TOP5 比分；含轻微泄漏，与复审 64.8%/31.5% 口径一致可比）；naiveRolling＝该场之前场次的最常见 3 档/TOP5（干净无泄漏）。生死线：修正矩阵 walk-forward 超不过 naiveRolling，引擎无上线理由（复审拍板）。

4. 口径诚实标注

对照统计主表为 in-sample（λ_DC 全季拟合含泄漏）——与设计 62.4% 同口径才可比；报告 notes[] 必须写明。干净结论看 Task 6 的 walk-forward 单联赛线。

#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""goal_engine —— 双轨分化设计进球引擎轨 P0（docs/2026-09-01-dual-track-prediction-design.html 第五节）。

阶段0 = 手写规则乘子（08-30/31 人工系数形式化起点）：
  λ' = λ_DC × exp( β·ln(r_att) + β·ln(r_def) )，β=0.3（手写超参，待阶段1学习器替换+消融）
  特征全部从 odds_{lg}_{season}.json 场次序列滚动构造（只用该场之前数据，无时间错位泄漏）：
    r_att  = 队近5主场(客场)场均进球(收缩) / 联赛主(客)队侧场均进球
    r_def  = 对侧近5客场(主场)场均失球(收缩) / 联赛客(主)队侧场均
    开季降权：该队此前场次 <2 → 特征权重 ×0.3（样本不足回落裸 DC）
消融开关：att / def / shrink（--disable）。
对照统计口径：in-sample（λ_DC 全季拟合含泄漏，与设计 62.4% 同口径可比；报告 notes 必须标注）。
开发者：sszhang（P0 由 Claude 实施，2026-09-01）
"""
import json
import math
import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parent))
from common import ROOT  # noqa: E402

CACHE = ROOT / "engine" / "cache"
BETA = 0.3            # 阶段0 手写乘子强度（学习器替换对象）
PRIOR_K = 2.0         # 小样本向联赛均值收缩的先验场次
WINDOW = 5            # 滚动窗口
EARLY_GAMES = 2       # 此前场次 <2 视为开季
EARLY_WEIGHT = 0.3

def pad_rate(team_sum, n, league_rate):
    """小样本收缩：(队累计 + 联赛均值×k) / (n+k)。"""
    return (team_sum + league_rate * PRIOR_K) / (n + PRIOR_K)

def league_sides(prior_matches):
    """联赛两侧进球基线：{"home_rate","away_rate"}（场均，用传入的（该场前）场次算）。"""
    n = len(prior_matches)
    if n == 0:
        return {"home_rate": 1.4, "away_rate": 1.1}   # 全局常识垫值（仅赛季头几场）
    hg = sum(m["fthg"] for m in prior_matches)
    ag = sum(m["ftag"] for m in prior_matches)
    return {"home_rate": hg / n, "away_rate": ag / n}

def build_history(prior_matches):
    """队级滚动账本：{"home_gf": {team:[...]}, "home_ga": ..., "away_gf": ..., "away_ga": ...}（进失球序列）。"""
    hist = {"home_gf": {}, "home_ga": {}, "away_gf": {}, "away_ga": {}}
    for m in prior_matches:
        hist["home_gf"].setdefault(m["home"], []).append(m["fthg"])
        hist["home_ga"].setdefault(m["home"], []).append(m["ftag"])
        hist["away_gf"].setdefault(m["away"], []).append(m["ftag"])
        hist["away_ga"].setdefault(m["away"], []).append(m["fthg"])
    return hist

def early_weight(games_before):
    return EARLY_WEIGHT if games_before < EARLY_GAMES else 1.0

def match_features(hist, sides, home, away):
    """返回 {"home_att","home_def","away_att","away_def","w_home","w_away"}——全部为 ratio（≥0）。"""
    def rate(book, team, side_rate):
        seq = hist[book].get(team, [])
        if not seq:
            return 1.0                                   # 无历史 → 不修正
        return pad_rate(sum(seq[-WINDOW:]), min(len(seq), WINDOW), side_rate)
    f = {
        "home_att": rate("home_gf", home, sides["home_rate"]) / sides["home_rate"],
        "home_def": rate("home_ga", home, sides["home_rate"]) / sides["home_rate"],   # 主队主场失球 vs 主侧均失
        "away_att": rate("away_gf", away, sides["away_rate"]) / sides["away_rate"],
        "away_def": rate("away_ga", away, sides["away_rate"]) / sides["away_rate"],
    }
    f["w_home"] = early_weight(len(hist["home_gf"].get(home, [])))
    f["w_away"] = early_weight(len(hist["away_gf"].get(away, [])))
    return f

注意 league_sides 用「该场之前」的场次聚合——对照统计主循环里逐场切片（见 T5），build_history 同理传前缀切片。home_def 分母用主队侧均值：失球基准=对侧产出，用同侧均值才同尺度。

import numpy as np
from dc_fit import dc_tau

def lambda_from_dc(dc, home, away):
    th, ta = dc["teams"].get(home), dc["teams"].get(away)
    if not th or not ta:
        return None
    return (math.exp(th["attack"] + ta["defense"] + dc["homeAdv"]),
            math.exp(ta["attack"] + th["defense"]))

def lambda_mult(f, side, disable=()):
    """side="home" → 用 home_att × away_def；side="away" 对称。开季权重乘进 β。"""
    att = f["home_att"] if side == "home" else f["away_att"]
    dfn = f["away_def"] if side == "home" else f["home_def"]
    w = f["w_home"] if side == "home" else f["w_away"]
    z = 0.0
    if "att" not in disable and att > 0:
        z += BETA * w * math.log(att)
    if "def" not in disable and dfn > 0:
        z += BETA * w * math.log(dfn)
    if "shrink" in disable:          # 消融：关开季降权 = 权重恒 1
        z /= w if w > 0 else 1.0
    return math.exp(z)

def score_matrix(lh, la, rho):
    """修正后 λ 重建 7×7 DC 比分矩阵（低分 tau 修正同 backtest.dc_three）。"""
    p = np.zeros((7, 7))
    for x in range(7):
        for y in range(7):
            pm = math.exp(-lh) * lh ** x / math.factorial(x) * math.exp(-la) * la ** y / math.factorial(y)
            p[x, y] = max(pm * dc_tau(x, y, lh, la, rho), 1e-12)
    p /= p.sum()
    return p

def ttg_probs(matrix):
    out = [0.0] * 14
    for i in range(7):
        for j in range(7):
            out[min(i + j, 13)] += matrix[i, j]
    return out

def crs_rank(matrix):
    return sorted(((f"{i}-{j}", float(matrix[i, j])) for i in range(7) for j in range(7)),
                  key=lambda kv: -kv[1])

def load_odds_matches(league, seasons=("2526",)):
    out = []
    for s in seasons:
        p = CACHE / f"odds_{league}_{s}.json"
        if not p.exists(): continue
        data = json.loads(p.read_text(encoding="utf-8"))
        for m in data.get("matches", []):
            if m.get("fthg") is None: continue
            try:
                out.append({"home": m["home"], "away": m["away"], "season": s,
                            "fthg": int(m["fthg"]), "ftag": int(m["ftag"])})
            except (ValueError, TypeError, KeyError): continue
    return out

def evaluate_league(league, disable=()):
    """逐场切片：prior = 该场之前的同联赛场次（跨季拼接按时间序）。
    四线统计：corrected / bareDc / naiveRolling / naiveStatic（static=全季最常见档，泄漏口径）。"""
    dc = json.loads((CACHE / f"{league}_dc.json").read_text(encoding="utf-8"))
    ms = load_odds_matches(league)
    counters = {k: {"ttg3": [0, 0], "crs1": [0, 0], "crs3": [0, 0], "crs5": [0, 0]} for k in
                ("corrected", "bareDc", "naiveRolling")}
    # naiveStatic：全季最常见 3 总进球档 + TOP5 比分（Counter，同复审口径）
    from collections import Counter
    tot_c = Counter(m["fthg"] + m["ftag"] for m in ms)
    scr_c = Counter(f"{m['fthg']}-{m['ftag']}" for m in ms)
    static_ttg3 = {g for g, _ in tot_c.most_common(3)}
    static_crs5 = {s for s, _ in scr_c.most_common(5)}
    n_static = {"ttg3": [0, 0], "crs1": [0, 0], "crs3": [0, 0], "crs5": [0, 0]}
    for idx, m in enumerate(ms):
        prior = ms[:idx]
        lam = lambda_from_dc(dc, m["home"], m["away"])
        if lam is None: continue
        actual_t = m["fthg"] + m["ftag"]; actual_s = f"{m['fthg']}-{m['ftag']}"
        sides = league_sides(prior); hist = build_history(prior)
        f = match_features(hist, sides, m["home"], m["away"])
        lh, la = lam
        lhc = lh * lambda_mult(f, "home", disable)
        lac = la * lambda_mult(f, "away", disable)
        for name, matrix in (("corrected", score_matrix(lhc, lac, dc["rho"])),
                             ("bareDc", score_matrix(lh, la, dc["rho"]))):
            bump(counters[name], matrix, actual_t, actual_s)
        # naiveRolling：prior 最常见 3 档 / TOP5 比分（prior 为空跳过该场）
        if prior:
            tc = Counter(x["fthg"] + x["ftag"] for x in prior)
            sc = Counter(f"{x['fthg']}-{x['ftag']}" for x in prior)
            r_ttg3 = {g for g, _ in tc.most_common(3)}
            r_crs5 = {s for s, _ in sc.most_common(5)}
            bump_naive(counters["naiveRolling"], r_ttg3, r_crs5, actual_t, actual_s)
        bump_naive(n_static, static_ttg3, static_crs5, actual_t, actual_s)
    return counters, n_static
